cmake_minimum_required(VERSION 3.24)
project(BadlionCosmeticPort VERSION 0.1.4 LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires Windows x64 MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(JDK_INCLUDE "C:/Program Files/Eclipse Adoptium/jdk-21.0.10.7-hotspot/include" CACHE PATH "JNI headers")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory(vendor/minhook)
add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/version-forward.exp
  COMMAND "${CMAKE_AR}" /nologo /machine:x64
    /def:${CMAKE_SOURCE_DIR}/src/version.def /out:${CMAKE_BINARY_DIR}/version-forward.lib
  DEPENDS src/version.def VERBATIM)
add_custom_target(forward-exports DEPENDS ${CMAKE_BINARY_DIR}/version-forward.exp)
foreach(target version version-test)
  add_library(${target} SHARED src/unlocker.cpp)
  add_dependencies(${target} forward-exports)
  target_include_directories(${target} PRIVATE src vendor ${JDK_INCLUDE} ${JDK_INCLUDE}/win32)
  target_link_libraries(${target} PRIVATE minhook ${CMAKE_BINARY_DIR}/version-forward.exp)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/patch_forwarders.py $<TARGET_FILE:${target}>)
endforeach()
target_compile_definitions(version-test PRIVATE BCP_TEST_BUILD)
add_library(badlion_electron SHARED tests/mock.cpp)
target_include_directories(badlion_electron PRIVATE ${JDK_INCLUDE} ${JDK_INCLUDE}/win32)
add_executable(harness tests/harness.cpp)
target_include_directories(harness PRIVATE src vendor ${JDK_INCLUDE} ${JDK_INCLUDE}/win32)
enable_testing()
configure_file(dist/cosmetics.json ${CMAKE_BINARY_DIR}/Release/cosmetics.json COPYONLY)
add_test(NAME native-hook COMMAND harness version-test.dll valid)
set_tests_properties(native-hook PROPERTIES WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/Release)
add_test(NAME version-guard COMMAND harness version.dll guard)
set_tests_properties(version-guard PROPERTIES WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/Release)
foreach(mode missing invalid diagnostic owned-export)
  file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/test-${mode})
  add_custom_command(TARGET harness POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:harness> $<TARGET_FILE:badlion_electron>
      $<TARGET_FILE:version-test> ${CMAKE_BINARY_DIR}/test-${mode})
  add_test(NAME catalog-${mode} COMMAND ${CMAKE_BINARY_DIR}/test-${mode}/harness.exe version-test.dll ${mode})
  set_tests_properties(catalog-${mode} PROPERTIES WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/test-${mode})
endforeach()
file(WRITE ${CMAKE_BINARY_DIR}/test-invalid/cosmetics.json "{broken json")
configure_file(dist/cosmetics.json ${CMAKE_BINARY_DIR}/test-diagnostic/cosmetics.json COPYONLY)
file(WRITE ${CMAKE_BINARY_DIR}/test-diagnostic/diagnostic-only "Preserve the stock response")
configure_file(dist/cosmetics.json ${CMAKE_BINARY_DIR}/test-owned-export/cosmetics.json COPYONLY)
file(WRITE ${CMAKE_BINARY_DIR}/test-owned-export/export-owned-to.txt "${CMAKE_BINARY_DIR}/test-owned-export/owned-cosmetics.json\n")
file(WRITE ${CMAKE_BINARY_DIR}/test-owned-export/mock-owned-response.json
  "{\"cosmetics\":[{\"cosmeticId\":2,\"cosmeticType\":\"CAPE\",\"active\":true,\"cosmeticData\":{},\"sourceData\":{},\"globalData\":{\"color\":123}}],\"privateRootField\":\"excluded\"}")
add_dependencies(harness badlion_electron version-test)
add_test(NAME java-version-skip COMMAND harness version-test.dll skip -version)
set_tests_properties(java-version-skip PROPERTIES WORKING_DIRECTORY ${CMAKE_BINARY_DIR}/Release)
set(GSON_JAR "$ENV{APPDATA}/.minecraft/libraries/com/google/code/gson/gson/2.2.4/gson-2.2.4.jar" CACHE FILEPATH "Client Gson jar for schema regression")
if(EXISTS "${GSON_JAR}")
  get_filename_component(JDK_ROOT "${JDK_INCLUDE}" DIRECTORY)
  add_custom_target(gson-schema-classes ALL
    COMMAND "${JDK_ROOT}/bin/javac.exe" -cp "${GSON_JAR}" -d ${CMAKE_BINARY_DIR}/gson-check
      ${CMAKE_SOURCE_DIR}/tests/GsonSchemaCheck.java VERBATIM)
  add_test(NAME client-gson-schema COMMAND "${JDK_ROOT}/bin/java.exe"
    -cp "${CMAKE_BINARY_DIR}/gson-check\;${GSON_JAR}" GsonSchemaCheck
    ${CMAKE_BINARY_DIR}/Release/last-payload.json)
  set_tests_properties(client-gson-schema PROPERTIES DEPENDS native-hook)
